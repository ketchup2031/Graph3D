cmake_minimum_required(VERSION 3.24)
project(MechForge LANGUAGES CXX)

find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
if(Qt6_VERSION VERSION_GREATER_EQUAL 7)
    message(FATAL_ERROR "MechForge requires Qt 6")
endif()
find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)
if(OpenCASCADE_MAJOR_VERSION LESS 7 OR OpenCASCADE_MAJOR_VERSION GREATER_EQUAL 8)
    message(FATAL_ERROR "MechForge requires Open CASCADE Technology 7.x")
endif()

add_library(mechforge_geometry STATIC src/geometry/PreviewBox.cpp)
target_compile_features(mechforge_geometry PUBLIC cxx_std_20)
set_target_properties(mechforge_geometry PROPERTIES CXX_EXTENSIONS OFF)
if(WIN32)
    # OCCT 7 headers use M_PI and related constants from the Windows CRT.
    target_compile_definitions(mechforge_geometry PRIVATE _USE_MATH_DEFINES)
endif()
target_include_directories(mechforge_geometry PUBLIC src)
target_include_directories(mechforge_geometry SYSTEM PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
target_link_libraries(mechforge_geometry PRIVATE TKPrim TKTopAlgo TKBRep TKMath TKernel)

add_executable(MechForge WIN32 src/app/main.cpp)
set_target_properties(MechForge PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(MechForge PRIVATE Qt6::Widgets mechforge_geometry)

include(CTest)
if(BUILD_TESTING)
    add_executable(preview_box_test tests/geometry/PreviewBoxTest.cpp)
    set_target_properties(preview_box_test PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries(preview_box_test PRIVATE mechforge_geometry)
    add_test(NAME preview_box COMMAND preview_box_test)
endif()
